cmake_minimum_required(VERSION 3.20)
list(APPEND BOARD_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(thinknode_m1 LANGUAGES C CXX)
include(../../cmake/common.cmake)
target_sources(app PRIVATE platform.cpp ${M1_COMMON} ${RADIO_SOURCES})

target_compile_definitions(app PRIVATE ${RADIO_DEFINES} M1_RTOS="zephyr" M1_VERSION="${M1_VERSION}")
target_compile_options(app PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions;-fno-rtti>)
